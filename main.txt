-- CosmiUI (same library used by Archivist Hub)
local CosmiUrl = "https://archivist-hub.archivisthub.workers.dev/cosmi-ui.lua"
local CosmiCache = "echo_cosmi_cache_v1.lua"

local function LoadCosmiSource()
    local Ok, Result = pcall(game.HttpGet, game, CosmiUrl)
    if Ok and type(Result) == "string" and #Result > 1000 then
        if writefile then
            pcall(writefile, CosmiCache, Result)
        end
        return Result
    end
    if isfile and readfile then
        local CacheOk, Cached = pcall(function()
            return isfile(CosmiCache) and readfile(CosmiCache)
        end)
        if CacheOk and type(Cached) == "string" and #Cached > 1000 then
            return Cached
        end
    end
    error("[Echo] CosmiUI download failed: " .. tostring(Result))
end

local function ResolveIcon(Name)
    if type(Name) ~= "string" then
        return Name
    end
    if Name == "custom" then
        return "rbxassetid://128673757580317"
    end
    return Name
end

WindUI = {

        if Settings.Notifications == false then

function Hub.register(Key, Element, Default)
    if Element ~= nil then
        table.insert(Hub.Registry, {Key = Key, Element = Element, Default = Default})
    end
end

local IdCounter = 0

local function NextId(TabName, Title)
    IdCounter += 1
    return (string.format("Echo_%s_%s_%d", TabName, tostring(Title or "Item"), IdCounter):gsub("%s+", ""))
end

local function WrapTab(Tab, TabName)
    local State = {Box = nil, Boxes = 0}

    local function NewBox(Name, Icon)
        State.Boxes += 1
        State.Box = Tab:AddGroupbox({
            Side = (State.Boxes % 2 == 1) and "Left" or "Right",
            Name = Name or TabName,
            IconName = ResolveIcon(Icon),
        })
        return State.Box
    end

    local function Box()
        return State.Box or NewBox(TabName)
    end

    local Wrapped = {Raw = Tab}

    function Wrapped.Section(_, Config)
        Config = Config or {}
        return NewBox(Config.Title or TabName, Config.Icon)
    end

    function Wrapped.Paragraph(_, Config)
        Config = Config or {}
        local Text = Config.Title or ""
        if Config.Desc and Config.Desc ~= "" then
            Text = (Text ~= "" and (Text .. "\n") or "") .. Config.Desc
        end
        return Box():AddLabel({Text = Text, DoesWrap = true})
    end

    function Wrapped.Toggle(_, Config)
        Config = Config or {}
        local Id = NextId(TabName, Config.Title)
        Box():AddToggle(Id, {
            Text = Config.Title or "Toggle",
            Default = Config.Value == true,
            Tooltip = Config.Desc,
            Callback = Config.Callback,
        })
        local Element = Library.Toggles[Id]
        if Config.Title then
            Hub.register(TabName .. "_" .. Config.Title, Element, Config.Value == true)
        end
        return Element
    end

    function Wrapped.Slider(_, Config)
        Config = Config or {}
        local Value = type(Config.Value) == "table" and Config.Value or {}
        local Step = Config.Step or 1
        local Min = Value.Min or 0
        local Id = NextId(TabName, Config.Title)
        Box():AddSlider(Id, {
            Text = Config.Title or "Slider",
            Default = Value.Default or Min,
            Min = Min,
            Max = Value.Max or 100,
            Rounding = (Step % 1 == 0) and 0 or 1,
            Tooltip = Config.Desc,
            Callback = Config.Callback,
        })
        local Element = Library.Options[Id]
        if Config.Title then
            Hub.register(TabName .. "_" .. Config.Title, Element, Value.Default or Min)
        end
        return Element
    end

    function Wrapped.Dropdown(_, Config)
        Config = Config or {}
        local Id = NextId(TabName, Config.Title)
        Box():AddDropdown(Id, {
            Text = Config.Title or "Dropdown",
            Values = Config.Values or {},
            Default = Config.Value,
            Multi = Config.Multi == true,
            Tooltip = Config.Desc,
            Callback = Config.Callback,
        })
        local Element = Library.Options[Id]
        if Config.Title then
            Hub.register(TabName .. "_" .. Config.Title, Element, Config.Value)
        end
        return Element
    end

    function Wrapped.Input(_, Config)
        Config = Config or {}
        local Id = NextId(TabName, Config.Title)
        Box():AddInput(Id, {
            Text = Config.Title or "Input",
            Default = Config.Value or "",
            Placeholder = Config.Placeholder or "",
            Tooltip = Config.Desc,
            Callback = Config.Callback,
        })
        local Element = Library.Options[Id]
        if Config.Title then
            Hub.register(TabName .. "_" .. Config.Title, Element, Config.Value or "")
        end
        return Element
    end

    function Wrapped.Button(_, Config)
        Config = Config or {}
        return Box():AddButton({
            Text = Config.Title or "Button",
            Tooltip = Config.Desc,
            Func = Config.Callback,
        })
    end

    function Wrapped.Divider(_)
        local Current = Box()
        if type(Current.AddDivider) == "function" then
            return Current:AddDivider()
        end
    end

    function Wrapped.Select(_)
        if type(Tab.Show) == "function" then
            return Tab:Show()
        end
    end

    return Wrapped
end

Hub.rawTab = function(Self, Config)
    local Tab = Self:AddTab({
        Name = Config.Title,
        Icon = ResolveIcon(Config.Icon) or "layout-dashboard",
        Description = Config.Desc or Config.Title,
    })
    return WrapTab(Tab, Config.Title or "Tab")
end

end
